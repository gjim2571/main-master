    20→export function setupInputListeners() {
    21→  const onKeyDown = (e: KeyboardEvent) => {
    22→    keys[e.code] = true;
    23→    if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Space', 'KeyW', 'KeyA', 'KeyS', 'KeyD'].includes(e.code)) {
    24→      e.preventDefault();
    25→    }
    26→  };
    27→  const onKeyUp = (e: KeyboardEvent) => {
    28→    keys[e.code] = false;
    29→  };
    30→  window.addEventListener('keydown', onKeyDown);
    31→  window.addEventListener('keyup', onKeyUp);
    32→  return () => {
    33→    window.removeEventListener('keydown', onKeyDown);
    34→    window.removeEventListener('keyup', onKeyUp);
    35→  };
    36→}

    38→function aabbCollision(ax: number, ay: number, aw: number, ah: number, bx: number, by: number, bw: number, bh: number) {
    39→  return ax < bx + bw && ax + aw > bx && ay < by + bh && ay + ah > by;
    40→}

    42→export function updateGame(state: GameState) {
    43→  if (state.phase !== 'playing') return;
    44→  if (!state.player.isAlive) return;
    45→
    46→  const { player } = state;
    47→  const dt = 1; // Fixed timestep for stability
    48→
    49→  // === Player Input ===
    50→  let moveX = 0;
    51→  if (keys['ArrowLeft'] || keys['KeyA']) moveX -= 1;
    52→  if (keys['ArrowRight'] || keys['KeyD']) moveX += 1;
    53→
    54→  player.vx = moveX * PLAYER_SPEED;
    55→  if (moveX !== 0) {
    56→    player.facing = moveX > 0 ? 'right' : 'left';
    57→  }
    58→
    59→  // Animation
    60→  if (Math.abs(player.vx) > 0.5) {
    61→    player.animTimer += dt;
    62→    if (player.animTimer > 6) {
    63→      player.animFrame = (player.animFrame + 1) % 4;
    64→      player.animTimer = 0;
    65→    }
    66→  } else {
    67→    player.animFrame = 0;
    68→    player.animTimer = 0;
    69→  }
    70→
    71→  // Jump (only on fresh press)
    72→  if ((keys['ArrowUp'] || keys['KeyW'] || keys['Space']) && player.jumpsLeft > 0) {
    73→    if (!keys._jumpHeld) {
    74→      player.vy = JUMP_FORCE;
    75→      player.jumpsLeft--;
    76→      keys._jumpHeld = true;
    77→      spawnParticles(state, player.x + player.width / 2, player.y + player.height, COLORS.neonGreen, 6, 2);
    78→    }
    79→  }
    80→  if (!keys['ArrowUp'] && !keys['KeyW'] && !keys['Space']) {
    81→    keys._jumpHeld = false;
    82→  }
    83→
    84→  // === Physics ===
    85→  player.vy += GRAVITY;
    86→  if (player.vy > MAX_FALL_SPEED) player.vy = MAX_FALL_SPEED;
    87→
    88→  // Move player
    89→  player.x += player.vx;
    90→  player.y += player.vy;
    91→
    92→  // Keep player in bounds horizontally
    93→  if (player.x < 0) player.x = 0;
    94→  if (player.x > LEVEL_WIDTH - player.width) player.x = LEVEL_WIDTH - player.width;
    95→
    96→  // Invincibility timer
    97→  if (player.invincible > 0) player.invincible -= 1;
    98→
    99→  // === Update Platforms ===
   100→  for (const plat of state.platforms) {
   101→    if (plat.y < -500) continue;
   102→    if (plat.type === 'moving' && plat.originX !== undefined && plat.moveRange !== undefined && plat.moveSpeed !== undefined) {
   103→      plat.x += plat.moveDir * plat.moveSpeed;
   104→      if (plat.x > plat.originX + plat.moveRange || plat.x < plat.originX - plat.moveRange) {
   105→        plat.moveDir *= -1;
   106→      }
   107→    }
   108→    if (plat.type === 'fragile' && plat.fragileTimer !== undefined) {
   109→      if (player.onGround && aabbCollision(player.x, player.y, player.width, player.height, plat.x, plat.y, plat.width, plat.height)) {
   110→        plat.fragileTimer -= 1;
   111→        // Flash warning
   112→      }
   113→      if (plat.fragileTimer <= 0) {
   114→        spawnParticles(state, plat.x + plat.width / 2, plat.y + plat.height / 2, COLORS.neonPink, 15, 4);
   115→        plat.y = -1000;
   116→      }
   117→    }
   118→  }
   119→
   120→  // === Platform Collisions (top-only, one-way) ===
   121→  player.onGround = false;
   122→  for (const plat of state.platforms) {
   123→    if (plat.y < -500) continue;
   124→
   125→    const playerBottom = player.y + player.height;
   126→    const playerRight = player.x + player.width;
   127→
   128→    // Check horizontal overlap first
   129→    if (playerRight <= plat.x || player.x >= plat.x + plat.width) continue;
   130→
   131→    // Check if player is falling and would land on this platform
   132→    if (player.vy >= 0 && playerBottom >= plat.y && playerBottom <= plat.y + plat.height + player.vy + 2) {
   133→      // Land on top
   134→      player.y = plat.y - player.height;
   135→      player.onGround = true;
   136→      player.jumpsLeft = player.maxJumps;
   137→
   138→      if (plat.type === 'bounce' && plat.bounceForce) {
   139→        player.vy = plat.bounceForce;
   140→        player.jumpsLeft = player.maxJumps;
   141→        spawnParticles(state, player.x + player.width / 2, plat.y, COLORS.neonGreen, 8, 3);
   142→      } else {
   143→        player.vy = 0;
   144→      }
   145→
   146→      // Moving platform carries player
   147→      if (plat.type === 'moving' && plat.moveSpeed !== undefined) {
   148→        player.x += plat.moveDir * plat.moveSpeed;
   149→      }
   150→    }
   151→  }
   152→
   153→  // === Coin Collection ===
   154→  for (const coin of state.coins) {
   155→    if (coin.collected) continue;
   156→
   157→    const cy = coin.y + Math.sin(Date.now() / 500 + coin.bobOffset) * 4;
   158→    const dx = (player.x + player.width / 2) - coin.x;
   159→    const dy = (player.y + player.height / 2) - cy;
   160→    const dist = Math.sqrt(dx * dx + dy * dy);
   161→
   162→    if (dist < coin.radius + 20) {
   163→      coin.collected = true;
   164→      state.score += coin.value;
   165→      const particleColor =
   166→        coin.type === 'legendary' ? COLORS.legendaryCoin : coin.type === 'rare' ? COLORS.rareCoin : COLORS.coin;
   167→      spawnParticles(state, coin.x, coin.y, particleColor, coin.type === 'legendary' ? 15 : 8, 4);
   168→    }
   169→  }
   170→
   171→  // === Enemy Update ===
   172→  for (const enemy of state.enemies) {
   173→    if (!enemy.isAlive) continue;
   174→
   175→    enemy.x += enemy.vx;
   176→    enemy.animTimer += 1;
   177→    if (enemy.animTimer > 10) {
   178→      enemy.animFrame = (enemy.animFrame + 1) % 2;
   179→      enemy.animTimer = 0;
   180→    }
   181→
   182→    if (enemy.type === 'flyer') {
   183→      enemy.y += Math.sin(Date.now() / 400) * 0.5;
   184→    }
   185→
   186→    if (enemy.x < enemy.originX || enemy.x > enemy.originX + enemy.moveRange) {
   187→      enemy.vx *= -1;
   188→    }
   189→
   190→    // Collision with player
   191→    if (player.invincible <= 0 && aabbCollision(player.x, player.y, player.width, player.height, enemy.x, enemy.y, enemy.width, enemy.height)) {
   192→      if (player.vy > 0 && player.y + player.height < enemy.y + enemy.height / 2 + 8) {
   193→        // Stomp
   194→        enemy.isAlive = false;
   195→        player.vy = JUMP_FORCE * 0.65;
   196→        state.score += 50;
   197→        spawnParticles(state, enemy.x + enemy.width / 2, enemy.y + enemy.height / 2, COLORS.neonPink, 12, 4);
   198→      } else {
   199→        // Take damage
   200→        player.isAlive = false;
   201→        state.phase = 'gameover';
   202→        spawnParticles(state, player.x + player.width / 2, player.y + player.height / 2, COLORS.neonPink, 20, 6);
   203→      }
   204→    }
   205→  }
   206→
   207→  // === Particles ===
   208→  for (let i = state.particles.length - 1; i >= 0; i--) {
   209→    const p = state.particles[i];
   210→    p.x += p.vx;
   211→    p.y += p.vy;
   212→    p.vy += 0.05;
   213→    p.life -= 1;
   214→    if (p.life <= 0) {
   215→      state.particles.splice(i, 1);
   216→    }
   217→  }
   218→
   219→  // === Camera (smooth follow) ===
   220→  const targetCameraX = player.x - CANVAS_WIDTH / 3;
   221→  state.camera.x += (targetCameraX - state.camera.x) * 0.08;
   222→  state.camera.x = Math.max(0, Math.min(LEVEL_WIDTH - CANVAS_WIDTH, state.camera.x));
   223→
   224→  state.backgroundOffset = state.camera.x * 0.3;
   225→  state.distance = Math.floor(player.x / 10);
   226→
   227→  // === Fall death ===
   228→  if (player.y > CANVAS_HEIGHT + 150) {
   229→    player.isAlive = false;
   230→    state.phase = 'gameover';
   231→  }
   232→}

   236→export function drawGame(ctx: CanvasRenderingContext2D, state: GameState, time: number, assets?: GameAssets) {
   237→  const { camera, player, platforms, coins, enemies, particles } = state;
   238→
   239→  // Clear + Background
   240→  ctx.fillStyle = COLORS.bg;
   241→  ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
   242→
   243→  const bgGrad = ctx.createLinearGradient(0, 0, 0, CANVAS_HEIGHT);
   244→  bgGrad.addColorStop(0, COLORS.bgGradient1);
   245→  bgGrad.addColorStop(1, COLORS.bgGradient2);
   246→  ctx.fillStyle = bgGrad;
   247→  ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
   248→
   249→  // Stars (parallax)
   250→  for (const star of state.stars) {
   251→    const sx = star.x - camera.x * star.speed;
   252→    const wrappedX = ((sx % (CANVAS_WIDTH + 100)) + CANVAS_WIDTH + 100) % (CANVAS_WIDTH + 100);
   253→    const alpha = star.brightness * (0.7 + 0.3 * Math.sin(time / 1000 + star.x));
   254→    ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
   255→    ctx.beginPath();
   256→    ctx.arc(wrappedX, star.y, star.size, 0, Math.PI * 2);
   257→    ctx.fill();
   258→  }
   259→
   260→  // Cityscape
   261→  drawCityscape(ctx, state);
   262→
   263→  // Camera transform for world objects
   264→  ctx.save();
   265→  ctx.translate(-camera.x, 0);
   266→
   267→  // === Floating Ritual Logo Arts ===
   268→  if (assets?.ritualLogoImg && assets.ritualLogoImg.complete) {
   269→    for (const art of state.floatingArts) {
   270→      const sx = art.x;
   271→      if (sx < camera.x - 100 || sx > camera.x + CANVAS_WIDTH + 100) continue;
   272→      const bobY = art.y + Math.sin(time / 1500 + art.bobOffset) * 10;
   273→      ctx.save();
   274→      ctx.globalAlpha = art.alpha;
   275→      ctx.translate(sx, bobY);
   276→      ctx.rotate(time * art.rotationSpeed);
   277→      ctx.drawImage(assets.ritualLogoImg, -art.size / 2, -art.size / 2, art.size, art.size);
   278→      ctx.restore();
   279→    }
   280→  }
   281→
   282→  // === Platforms ===
   283→  for (const plat of platforms) {
   284→    if (plat.y < -500) continue;
   285→    if (plat.x + plat.width < camera.x - 50 || plat.x > camera.x + CANVAS_WIDTH + 50) continue;
   286→
   287→    ctx.shadowColor = plat.glowColor || COLORS.platformBorder;
   288→    ctx.shadowBlur = 8;
   289→    ctx.fillStyle = plat.color || COLORS.platform;
   290→    ctx.fillRect(plat.x, plat.y, plat.width, plat.height);
   291→
   292→    ctx.strokeStyle = plat.glowColor || COLORS.platformBorder;
   293→    ctx.lineWidth = 1.5;
   294→    ctx.strokeRect(plat.x, plat.y, plat.width, plat.height);
   295→
   296→    // Top highlight
   297→    ctx.lineWidth = 2;
   298→    ctx.beginPath();
   299→    ctx.moveTo(plat.x, plat.y);
   300→    ctx.lineTo(plat.x + plat.width, plat.y);
   301→    ctx.stroke();
   302→    ctx.shadowBlur = 0;
   303→
   304→    // Fragile warning flash
   305→    if (plat.type === 'fragile' && plat.fragileTimer !== undefined && plat.fragileTimer < 40) {
   306→      const alpha = plat.fragileTimer < 20 ? 0.3 + 0.7 * Math.abs(Math.sin(time / 80)) : 0.3;
   307→      ctx.fillStyle = `rgba(255, 51, 102, ${alpha})`;
   308→      ctx.fillRect(plat.x, plat.y, plat.width, plat.height);
   309→    }
   310→  }
   311→
   312→  // === Coins ===
   313→  for (const coin of state.coins) {
   314→    if (coin.collected) continue;
   315→
   316→    const cy = coin.y + Math.sin(time / 500 + coin.bobOffset) * 4;
   317→    const color = coin.type === 'legendary' ? COLORS.legendaryCoin : coin.type === 'rare' ? COLORS.rareCoin : COLORS.coin;
   318→
   319→    ctx.shadowColor = color;
   320→    ctx.shadowBlur = coin.type === 'legendary' ? 15 : 10;
   321→    ctx.fillStyle = color;
   322→    ctx.beginPath();
   323→    ctx.arc(coin.x, cy, coin.radius, 0, Math.PI * 2);
   324→    ctx.fill();
   325→
   326→    // Shine
   327→    ctx.fillStyle = 'rgba(255, 255, 255, 0.4)';
   328→    ctx.beginPath();
   329→    ctx.arc(coin.x - coin.radius * 0.2, cy - coin.radius * 0.2, coin.radius * 0.35, 0, Math.PI * 2);
   330→    ctx.fill();
   331→
   332→    // Legendary sparkle
   333→    if (coin.type === 'legendary') {
   334→      ctx.strokeStyle = COLORS.legendaryCoin;
   335→      ctx.lineWidth = 1;
   336→      for (let s = 0; s < 4; s++) {
   337→        const angle = time / 200 + (s * Math.PI) / 2;
   338→        const len = coin.radius + 5 + Math.sin(time / 150 + s) * 3;
   339→        ctx.beginPath();
   340→        ctx.moveTo(coin.x + Math.cos(angle) * coin.radius, cy + Math.sin(angle) * coin.radius);
   341→        ctx.lineTo(coin.x + Math.cos(angle) * len, cy + Math.sin(angle) * len);
   342→        ctx.stroke();
   343→      }
   344→    }
   345→    ctx.shadowBlur = 0;
   346→  }
   347→
   348→  // === Enemies ===
   349→  for (const enemy of state.enemies) {
   350→    if (!enemy.isAlive) continue;
   351→    ctx.shadowColor = COLORS.enemyGlow;
   352→    ctx.shadowBlur = 10;
   353→    ctx.fillStyle = COLORS.enemy;
   354→
   355→    if (enemy.type === 'flyer') {
   356→      ctx.beginPath();
   357→      ctx.moveTo(enemy.x + enemy.width / 2, enemy.y);
   358→      ctx.lineTo(enemy.x + enemy.width, enemy.y + enemy.height);
   359→      ctx.lineTo(enemy.x, enemy.y + enemy.height);
   360→      ctx.closePath();
   361→      ctx.fill();
   362→
   363→      const wingOffset = Math.sin(time / 150) * 3;
   364→      ctx.strokeStyle = COLORS.neonPink;
   365→      ctx.lineWidth = 2;
   366→      ctx.beginPath();
   367→      ctx.moveTo(enemy.x, enemy.y + enemy.height * 0.4);
   368→      ctx.lineTo(enemy.x - 8, enemy.y + wingOffset);
   369→      ctx.stroke();
   370→      ctx.beginPath();
   371→      ctx.moveTo(enemy.x + enemy.width, enemy.y + enemy.height * 0.4);
   372→      ctx.lineTo(enemy.x + enemy.width + 8, enemy.y + wingOffset);
   373→      ctx.stroke();
   374→    } else {
   375→      ctx.fillRect(enemy.x, enemy.y, enemy.width, enemy.height);
   376→      ctx.fillStyle = '#ffffff';
   377→      const eyeOff = enemy.vx > 0 ? 4 : -1;
   378→      ctx.fillRect(enemy.x + enemy.width / 2 + eyeOff - 2, enemy.y + 6, 3, 3);
   379→      ctx.fillRect(enemy.x + enemy.width / 2 + eyeOff + 4, enemy.y + 6, 3, 3);
   380→    }
   381→    ctx.shadowBlur = 0;
   382→  }
   383→
   384→  // === Player ===
   385→  if (player.isAlive) {
   386→    const px = player.x;
   387→    const py = player.y;
   388→    const pw = player.width;
   389→    const ph = player.height;
   390→
   391→    // Trail
   392→    if (Math.abs(player.vx) > 1 || !player.onGround) {
   393→      ctx.fillStyle = COLORS.playerTrail;
   394→      ctx.beginPath();
   395→      ctx.roundRect(px - 2, py + 4, pw, ph - 4, 4);
   396→      ctx.fill();
   397→    }
   398→
   399→    if (assets?.characterImg && assets.characterImg.complete) {
   400→      // Character image sprite
   401→      ctx.save();
   402→      ctx.shadowColor = COLORS.playerGlow;
   403→      ctx.shadowBlur = 12;
   404→
   405→      if (player.facing === 'left') {
   406→        ctx.translate(px + pw / 2, 0);
   407→        ctx.scale(-1, 1);
   408→        ctx.drawImage(assets.characterImg, -pw / 2 - 2, py - 2, pw + 4, ph + 4);
   409→      } else {
   410→        ctx.drawImage(assets.characterImg, px - 2, py - 2, pw + 4, ph + 4);
   411→      }
   412→
   413→      // Lightning aura
   414→      const pulseAlpha = 0.12 + 0.08 * Math.sin(time / 300);
   415→      ctx.strokeStyle = `rgba(0, 255, 170, ${pulseAlpha})`;
   416→      ctx.lineWidth = 2;
   417→      ctx.beginPath();
   418→      ctx.roundRect(px - 4, py - 4, pw + 8, ph + 8, 8);
   419→      ctx.stroke();
   420→      ctx.restore();
   421→    } else {
   422→      // Fallback geometric player
   423→      ctx.shadowColor = COLORS.playerGlow;
   424→      ctx.shadowBlur = 12;
   425→      ctx.fillStyle = COLORS.player;
   426→      ctx.beginPath();
   427→      ctx.roundRect(px, py, pw, ph, 5);
   428→      ctx.fill();
   429→
   430→      const faceDir = player.facing === 'right' ? 1 : -1;
   431→      ctx.fillStyle = '#0a0a2e';
   432→      ctx.beginPath();
   433→      ctx.arc(px + pw / 2 + faceDir * 4, py + 10, 3, 0, Math.PI * 2);
   434→      ctx.fill();
   435→      ctx.fillStyle = '#ffffff';
   436→      ctx.beginPath();
   437→      ctx.arc(px + pw / 2 + faceDir * 5, py + 9, 1, 0, Math.PI * 2);
   438→      ctx.fill();
   439→      ctx.shadowBlur = 0;
   440→    }
   441→
   442→    // Legs animation
   443→    ctx.fillStyle = COLORS.player;
   444→    if (player.onGround && Math.abs(player.vx) > 0.5) {
   445→      const legOff = Math.sin(time / 80) * 4;
   446→      ctx.fillRect(px + 7, py + ph, 4, 4 + legOff);
   447→      ctx.fillRect(px + pw - 11, py + ph, 4, 4 - legOff);
   448→    } else {
   449→      ctx.fillRect(px + 7, py + ph, 4, 3);
   450→      ctx.fillRect(px + pw - 11, py + ph, 4, 3);
   451→    }
   452→
   453→    // Jetpack flame
   454→    if (!player.onGround) {
   455→      ctx.fillStyle = COLORS.neonBlue;
   456→      ctx.fillRect(px + pw / 2 - 4, py + ph + 2, 8, 3);
   457→      const flameLen = 5 + Math.random() * 5;
   458→      ctx.fillStyle = COLORS.neonYellow;
   459→      ctx.beginPath();
   460→      ctx.moveTo(px + pw / 2 - 3, py + ph + 5);
   461→      ctx.lineTo(px + pw / 2 + 3, py + ph + 5);
   462→      ctx.lineTo(px + pw / 2, py + ph + 5 + flameLen);
   463→      ctx.closePath();
   464→      ctx.fill();
   465→    }
   466→    ctx.shadowBlur = 0;
   467→  }
   468→
   469→  // === Particles ===
   470→  for (const p of particles) {
   471→    const alpha = p.life / p.maxLife;
   472→    ctx.globalAlpha = alpha;
   473→    ctx.fillStyle = p.color;
   474→    ctx.beginPath();
   475→    ctx.arc(p.x, p.y, p.size * alpha, 0, Math.PI * 2);
   476→    ctx.fill();
   477→  }
   478→  ctx.globalAlpha = 1;
   479→
   480→  ctx.restore();
   481→}

   483→function drawCityscape(ctx: CanvasRenderingContext2D, state: GameState) {
   484→  const offset = state.camera.x * 0.15;
   485→
   486→  ctx.fillStyle = 'rgba(15, 15, 45, 0.8)';
   487→  const bw = [40, 30, 50, 35, 45, 25, 55, 30, 40, 60];
   488→  for (let i = 0; i < bw.length; i++) {
   489→    const bx = i * 90 - (offset % 90);
   490→    const bh = 80 + bw[i] + Math.sin(i * 2.5) * 30;
   491→    ctx.fillRect(bx, CANVAS_HEIGHT - bh, bw[i], bh);
   492→
   493→    ctx.fillStyle = 'rgba(0, 255, 170, 0.15)';
   494→    for (let wy = CANVAS_HEIGHT - bh + 10; wy < CANVAS_HEIGHT - 20; wy += 15) {
   495→      for (let wx = bx + 5; wx < bx + bw[i] - 5; wx += 10) {
   496→        if (Math.sin(wx * 7 + wy * 3 + i) > 0) {
   497→          ctx.fillRect(wx, wy, 4, 6);
   498→        }
   499→      }
   500→    }
   501→    ctx.fillStyle = 'rgba(15, 15, 45, 0.8)';
   502→  }
   503→
   504→  ctx.strokeStyle = 'rgba(0, 255, 170, 0.2)';
   505→  ctx.lineWidth = 1;
   506→  ctx.beginPath();
   507→  ctx.moveTo(0, CANVAS_HEIGHT - 30);
   508→  ctx.lineTo(CANVAS_WIDTH, CANVAS_HEIGHT - 30);
   509→  ctx.stroke();
   510→}

   512→function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   513→  ctx.beginPath();
   514→  ctx.moveTo(x + r, y);
   515→  ctx.lineTo(x + w - r, y);
   516→  ctx.quadraticCurveTo(x + w, y, x + w, y + r);
   517→  ctx.lineTo(x + w, y + h - r);
   518→  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
   519→  ctx.lineTo(x + r, y + h);
   520→  ctx.quadraticCurveTo(x, y + h, x, y + h - r);
   521→  ctx.lineTo(x, y + r);
   522→  ctx.quadraticCurveTo(x, y, x + r, y);
   523→  ctx.closePath();
   524→}
