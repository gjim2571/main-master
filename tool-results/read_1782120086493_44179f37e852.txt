    20→    20→export function setupInputListeners() {
    21→    21→  const onKeyDown = (e: KeyboardEvent) => {
    22→    22→    keys[e.code] = true;
    23→    23→    if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Space', 'KeyW', 'KeyA', 'KeyS', 'KeyD'].includes(e.code)) {
    24→    24→      e.preventDefault();
    25→    25→    }
    26→    26→  };
    27→    27→  const onKeyUp = (e: KeyboardEvent) => {
    28→    28→    keys[e.code] = false;
    29→    29→  };
    30→    30→  window.addEventListener('keydown', onKeyDown);
    31→    31→  window.addEventListener('keyup', onKeyUp);
    32→    32→  return () => {
    33→    33→    window.removeEventListener('keydown', onKeyDown);
    34→    34→    window.removeEventListener('keyup', onKeyUp);
    35→    35→  };
    36→    36→}

    38→    38→function aabbCollision(ax: number, ay: number, aw: number, ah: number, bx: number, by: number, bw: number, bh: number) {
    39→    39→  return ax < bx + bw && ax + aw > bx && ay < by + bh && ay + ah > by;
    40→    40→}

    42→    42→export function updateGame(state: GameState) {
    43→    43→  if (state.phase !== 'playing') return;
    44→    44→  if (!state.player.isAlive) return;
    45→    45→
    46→    46→  const { player } = state;
    47→    47→  const dt = 1; // Fixed timestep for stability
    48→    48→
    49→    49→  // === Player Input ===
    50→    50→  let moveX = 0;
    51→    51→  if (keys['ArrowLeft'] || keys['KeyA']) moveX -= 1;
    52→    52→  if (keys['ArrowRight'] || keys['KeyD']) moveX += 1;
    53→    53→
    54→    54→  player.vx = moveX * PLAYER_SPEED;
    55→    55→  if (moveX !== 0) {
    56→    56→    player.facing = moveX > 0 ? 'right' : 'left';
    57→    57→  }
    58→    58→
    59→    59→  // Animation
    60→    60→  if (Math.abs(player.vx) > 0.5) {
    61→    61→    player.animTimer += dt;
    62→    62→    if (player.animTimer > 6) {
    63→    63→      player.animFrame = (player.animFrame + 1) % 4;
    64→    64→      player.animTimer = 0;
    65→    65→    }
    66→    66→  } else {
    67→    67→    player.animFrame = 0;
    68→    68→    player.animTimer = 0;
    69→    69→  }
    70→    70→
    71→    71→  // Jump (only on fresh press)
    72→    72→  if ((keys['ArrowUp'] || keys['KeyW'] || keys['Space']) && player.jumpsLeft > 0) {
    73→    73→    if (!keys._jumpHeld) {
    74→    74→      player.vy = JUMP_FORCE;
    75→    75→      player.jumpsLeft--;
    76→    76→      keys._jumpHeld = true;
    77→    77→      spawnParticles(state, player.x + player.width / 2, player.y + player.height, COLORS.neonGreen, 6, 2);
    78→    78→    }
    79→    79→  }
    80→    80→  if (!keys['ArrowUp'] && !keys['KeyW'] && !keys['Space']) {
    81→    81→    keys._jumpHeld = false;
    82→    82→  }
    83→    83→
    84→    84→  // === Physics ===
    85→    85→  player.vy += GRAVITY;
    86→    86→  if (player.vy > MAX_FALL_SPEED) player.vy = MAX_FALL_SPEED;
    87→    87→
    88→    88→  // Move player
    89→    89→  player.x += player.vx;
    90→    90→  player.y += player.vy;
    91→    91→
    92→    92→  // Keep player in bounds horizontally
    93→    93→  if (player.x < 0) player.x = 0;
    94→    94→  if (player.x > LEVEL_WIDTH - player.width) player.x = LEVEL_WIDTH - player.width;
    95→    95→
    96→    96→  // Invincibility timer
    97→    97→  if (player.invincible > 0) player.invincible -= 1;
    98→    98→
    99→    99→  // === Update Platforms ===
   100→   100→  for (const plat of state.platforms) {
   101→   101→    if (plat.y < -500) continue;
   102→   102→    if (plat.type === 'moving' && plat.originX !== undefined && plat.moveRange !== undefined && plat.moveSpeed !== undefined) {
   103→   103→      plat.x += plat.moveDir * plat.moveSpeed;
   104→   104→      if (plat.x > plat.originX + plat.moveRange || plat.x < plat.originX - plat.moveRange) {
   105→   105→        plat.moveDir *= -1;
   106→   106→      }
   107→   107→    }
   108→   108→    if (plat.type === 'fragile' && plat.fragileTimer !== undefined) {
   109→   109→      if (player.onGround && aabbCollision(player.x, player.y, player.width, player.height, plat.x, plat.y, plat.width, plat.height)) {
   110→   110→        plat.fragileTimer -= 1;
   111→   111→        // Flash warning
   112→   112→      }
   113→   113→      if (plat.fragileTimer <= 0) {
   114→   114→        spawnParticles(state, plat.x + plat.width / 2, plat.y + plat.height / 2, COLORS.neonPink, 15, 4);
   115→   115→        plat.y = -1000;
   116→   116→      }
   117→   117→    }
   118→   118→  }
   119→   119→
   120→   120→  // === Platform Collisions (top-only, one-way) ===
   121→   121→  player.onGround = false;
   122→   122→  for (const plat of state.platforms) {
   123→   123→    if (plat.y < -500) continue;
   124→   124→
   125→   125→    const playerBottom = player.y + player.height;
   126→   126→    const playerRight = player.x + player.width;
   127→   127→
   128→   128→    // Check horizontal overlap first
   129→   129→    if (playerRight <= plat.x || player.x >= plat.x + plat.width) continue;
   130→   130→
   131→   131→    // Check if player is falling and would land on this platform
   132→   132→    if (player.vy >= 0 && playerBottom >= plat.y && playerBottom <= plat.y + plat.height + player.vy + 2) {
   133→   133→      // Land on top
   134→   134→      player.y = plat.y - player.height;
   135→   135→      player.onGround = true;
   136→   136→      player.jumpsLeft = player.maxJumps;
   137→   137→
   138→   138→      if (plat.type === 'bounce' && plat.bounceForce) {
   139→   139→        player.vy = plat.bounceForce;
   140→   140→        player.jumpsLeft = player.maxJumps;
   141→   141→        spawnParticles(state, player.x + player.width / 2, plat.y, COLORS.neonGreen, 8, 3);
   142→   142→      } else {
   143→   143→        player.vy = 0;
   144→   144→      }
   145→   145→
   146→   146→      // Moving platform carries player
   147→   147→      if (plat.type === 'moving' && plat.moveSpeed !== undefined) {
   148→   148→        player.x += plat.moveDir * plat.moveSpeed;
   149→   149→      }
   150→   150→    }
   151→   151→  }
   152→   152→
   153→   153→  // === Coin Collection ===
   154→   154→  for (const coin of state.coins) {
   155→   155→    if (coin.collected) continue;
   156→   156→
   157→   157→    const cy = coin.y + Math.sin(Date.now() / 500 + coin.bobOffset) * 4;
   158→   158→    const dx = (player.x + player.width / 2) - coin.x;
   159→   159→    const dy = (player.y + player.height / 2) - cy;
   160→   160→    const dist = Math.sqrt(dx * dx + dy * dy);
   161→   161→
   162→   162→    if (dist < coin.radius + 20) {
   163→   163→      coin.collected = true;
   164→   164→      state.score += coin.value;
   165→   165→      const particleColor =
   166→   166→        coin.type === 'legendary' ? COLORS.legendaryCoin : coin.type === 'rare' ? COLORS.rareCoin : COLORS.coin;
   167→   167→      spawnParticles(state, coin.x, coin.y, particleColor, coin.type === 'legendary' ? 15 : 8, 4);
   168→   168→    }
   169→   169→  }
   170→   170→
   171→   171→  // === Enemy Update ===
   172→   172→  for (const enemy of state.enemies) {
   173→   173→    if (!enemy.isAlive) continue;
   174→   174→
   175→   175→    enemy.x += enemy.vx;
   176→   176→    enemy.animTimer += 1;
   177→   177→    if (enemy.animTimer > 10) {
   178→   178→      enemy.animFrame = (enemy.animFrame + 1) % 2;
   179→   179→      enemy.animTimer = 0;
   180→   180→    }
   181→   181→
   182→   182→    if (enemy.type === 'flyer') {
   183→   183→      enemy.y += Math.sin(Date.now() / 400) * 0.5;
   184→   184→    }
   185→   185→
   186→   186→    if (enemy.x < enemy.originX || enemy.x > enemy.originX + enemy.moveRange) {
   187→   187→      enemy.vx *= -1;
   188→   188→    }
   189→   189→
   190→   190→    // Collision with player
   191→   191→    if (player.invincible <= 0 && aabbCollision(player.x, player.y, player.width, player.height, enemy.x, enemy.y, enemy.width, enemy.height)) {
   192→   192→      if (player.vy > 0 && player.y + player.height < enemy.y + enemy.height / 2 + 8) {
   193→   193→        // Stomp
   194→   194→        enemy.isAlive = false;
   195→   195→        player.vy = JUMP_FORCE * 0.65;
   196→   196→        state.score += 50;
   197→   197→        spawnParticles(state, enemy.x + enemy.width / 2, enemy.y + enemy.height / 2, COLORS.neonPink, 12, 4);
   198→   198→      } else {
   199→   199→        // Take damage
   200→   200→        player.isAlive = false;
   201→   201→        state.phase = 'gameover';
   202→   202→        spawnParticles(state, player.x + player.width / 2, player.y + player.height / 2, COLORS.neonPink, 20, 6);
   203→   203→      }
   204→   204→    }
   205→   205→  }
   206→   206→
   207→   207→  // === Particles ===
   208→   208→  for (let i = state.particles.length - 1; i >= 0; i--) {
   209→   209→    const p = state.particles[i];
   210→   210→    p.x += p.vx;
   211→   211→    p.y += p.vy;
   212→   212→    p.vy += 0.05;
   213→   213→    p.life -= 1;
   214→   214→    if (p.life <= 0) {
   215→   215→      state.particles.splice(i, 1);
   216→   216→    }
   217→   217→  }
   218→   218→
   219→   219→  // === Camera (smooth follow) ===
   220→   220→  const targetCameraX = player.x - CANVAS_WIDTH / 3;
   221→   221→  state.camera.x += (targetCameraX - state.camera.x) * 0.08;
   222→   222→  state.camera.x = Math.max(0, Math.min(LEVEL_WIDTH - CANVAS_WIDTH, state.camera.x));
   223→   223→
   224→   224→  state.backgroundOffset = state.camera.x * 0.3;
   225→   225→  state.distance = Math.floor(player.x / 10);
   226→   226→
   227→   227→  // === Fall death ===
   228→   228→  if (player.y > CANVAS_HEIGHT + 150) {
   229→   229→    player.isAlive = false;
   230→   230→    state.phase = 'gameover';
   231→   231→  }
   232→   232→}

   236→   236→export function drawGame(ctx: CanvasRenderingContext2D, state: GameState, time: number, assets?: GameAssets) {
   237→   237→  const { camera, player, platforms, coins, enemies, particles } = state;
   238→   238→
   239→   239→  // Clear + Background
   240→   240→  ctx.fillStyle = COLORS.bg;
   241→   241→  ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
   242→   242→
   243→   243→  const bgGrad = ctx.createLinearGradient(0, 0, 0, CANVAS_HEIGHT);
   244→   244→  bgGrad.addColorStop(0, COLORS.bgGradient1);
   245→   245→  bgGrad.addColorStop(1, COLORS.bgGradient2);
   246→   246→  ctx.fillStyle = bgGrad;
   247→   247→  ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
   248→   248→
   249→   249→  // Stars (parallax)
   250→   250→  for (const star of state.stars) {
   251→   251→    const sx = star.x - camera.x * star.speed;
   252→   252→    const wrappedX = ((sx % (CANVAS_WIDTH + 100)) + CANVAS_WIDTH + 100) % (CANVAS_WIDTH + 100);
   253→   253→    const alpha = star.brightness * (0.7 + 0.3 * Math.sin(time / 1000 + star.x));
   254→   254→    ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
   255→   255→    ctx.beginPath();
   256→   256→    ctx.arc(wrappedX, star.y, star.size, 0, Math.PI * 2);
   257→   257→    ctx.fill();
   258→   258→  }
   259→   259→
   260→   260→  // Cityscape
   261→   261→  drawCityscape(ctx, state);
   262→   262→
   263→   263→  // Camera transform for world objects
   264→   264→  ctx.save();
   265→   265→  ctx.translate(-camera.x, 0);
   266→   266→
   267→   267→  // === Floating Ritual Logo Arts ===
   268→   268→  if (assets?.ritualLogoImg && assets.ritualLogoImg.complete) {
   269→   269→    for (const art of state.floatingArts) {
   270→   270→      const sx = art.x;
   271→   271→      if (sx < camera.x - 100 || sx > camera.x + CANVAS_WIDTH + 100) continue;
   272→   272→      const bobY = art.y + Math.sin(time / 1500 + art.bobOffset) * 10;
   273→   273→      ctx.save();
   274→   274→      ctx.globalAlpha = art.alpha;
   275→   275→      ctx.translate(sx, bobY);
   276→   276→      ctx.rotate(time * art.rotationSpeed);
   277→   277→      ctx.drawImage(assets.ritualLogoImg, -art.size / 2, -art.size / 2, art.size, art.size);
   278→   278→      ctx.restore();
   279→   279→    }
   280→   280→  }
   281→   281→
   282→   282→  // === Platforms ===
   283→   283→  for (const plat of platforms) {
   284→   284→    if (plat.y < -500) continue;
   285→   285→    if (plat.x + plat.width < camera.x - 50 || plat.x > camera.x + CANVAS_WIDTH + 50) continue;
   286→   286→
   287→   287→    ctx.shadowColor = plat.glowColor || COLORS.platformBorder;
   288→   288→    ctx.shadowBlur = 8;
   289→   289→    ctx.fillStyle = plat.color || COLORS.platform;
   290→   290→    ctx.fillRect(plat.x, plat.y, plat.width, plat.height);
   291→   291→
   292→   292→    ctx.strokeStyle = plat.glowColor || COLORS.platformBorder;
   293→   293→    ctx.lineWidth = 1.5;
   294→   294→    ctx.strokeRect(plat.x, plat.y, plat.width, plat.height);
   295→   295→
   296→   296→    // Top highlight
   297→   297→    ctx.lineWidth = 2;
   298→   298→    ctx.beginPath();
   299→   299→    ctx.moveTo(plat.x, plat.y);
   300→   300→    ctx.lineTo(plat.x + plat.width, plat.y);
   301→   301→    ctx.stroke();
   302→   302→    ctx.shadowBlur = 0;
   303→   303→
   304→   304→    // Fragile warning flash
   305→   305→    if (plat.type === 'fragile' && plat.fragileTimer !== undefined && plat.fragileTimer < 40) {
   306→   306→      const alpha = plat.fragileTimer < 20 ? 0.3 + 0.7 * Math.abs(Math.sin(time / 80)) : 0.3;
   307→   307→      ctx.fillStyle = `rgba(255, 51, 102, ${alpha})`;
   308→   308→      ctx.fillRect(plat.x, plat.y, plat.width, plat.height);
   309→   309→    }
   310→   310→  }
   311→   311→
   312→   312→  // === Coins ===
   313→   313→  for (const coin of state.coins) {
   314→   314→    if (coin.collected) continue;
   315→   315→
   316→   316→    const cy = coin.y + Math.sin(time / 500 + coin.bobOffset) * 4;
   317→   317→    const color = coin.type === 'legendary' ? COLORS.legendaryCoin : coin.type === 'rare' ? COLORS.rareCoin : COLORS.coin;
   318→   318→
   319→   319→    ctx.shadowColor = color;
   320→   320→    ctx.shadowBlur = coin.type === 'legendary' ? 15 : 10;
   321→   321→    ctx.fillStyle = color;
   322→   322→    ctx.beginPath();
   323→   323→    ctx.arc(coin.x, cy, coin.radius, 0, Math.PI * 2);
   324→   324→    ctx.fill();
   325→   325→
   326→   326→    // Shine
   327→   327→    ctx.fillStyle = 'rgba(255, 255, 255, 0.4)';
   328→   328→    ctx.beginPath();
   329→   329→    ctx.arc(coin.x - coin.radius * 0.2, cy - coin.radius * 0.2, coin.radius * 0.35, 0, Math.PI * 2);
   330→   330→    ctx.fill();
   331→   331→
   332→   332→    // Legendary sparkle
   333→   333→    if (coin.type === 'legendary') {
   334→   334→      ctx.strokeStyle = COLORS.legendaryCoin;
   335→   335→      ctx.lineWidth = 1;
   336→   336→      for (let s = 0; s < 4; s++) {
   337→   337→        const angle = time / 200 + (s * Math.PI) / 2;
   338→   338→        const len = coin.radius + 5 + Math.sin(time / 150 + s) * 3;
   339→   339→        ctx.beginPath();
   340→   340→        ctx.moveTo(coin.x + Math.cos(angle) * coin.radius, cy + Math.sin(angle) * coin.radius);
   341→   341→        ctx.lineTo(coin.x + Math.cos(angle) * len, cy + Math.sin(angle) * len);
   342→   342→        ctx.stroke();
   343→   343→      }
   344→   344→    }
   345→   345→    ctx.shadowBlur = 0;
   346→   346→  }
   347→   347→
   348→   348→  // === Enemies ===
   349→   349→  for (const enemy of state.enemies) {
   350→   350→    if (!enemy.isAlive) continue;
   351→   351→    ctx.shadowColor = COLORS.enemyGlow;
   352→   352→    ctx.shadowBlur = 10;
   353→   353→    ctx.fillStyle = COLORS.enemy;
   354→   354→
   355→   355→    if (enemy.type === 'flyer') {
   356→   356→      ctx.beginPath();
   357→   357→      ctx.moveTo(enemy.x + enemy.width / 2, enemy.y);
   358→   358→      ctx.lineTo(enemy.x + enemy.width, enemy.y + enemy.height);
   359→   359→      ctx.lineTo(enemy.x, enemy.y + enemy.height);
   360→   360→      ctx.closePath();
   361→   361→      ctx.fill();
   362→   362→
   363→   363→      const wingOffset = Math.sin(time / 150) * 3;
   364→   364→      ctx.strokeStyle = COLORS.neonPink;
   365→   365→      ctx.lineWidth = 2;
   366→   366→      ctx.beginPath();
   367→   367→      ctx.moveTo(enemy.x, enemy.y + enemy.height * 0.4);
   368→   368→      ctx.lineTo(enemy.x - 8, enemy.y + wingOffset);
   369→   369→      ctx.stroke();
   370→   370→      ctx.beginPath();
   371→   371→      ctx.moveTo(enemy.x + enemy.width, enemy.y + enemy.height * 0.4);
   372→   372→      ctx.lineTo(enemy.x + enemy.width + 8, enemy.y + wingOffset);
   373→   373→      ctx.stroke();
   374→   374→    } else {
   375→   375→      ctx.fillRect(enemy.x, enemy.y, enemy.width, enemy.height);
   376→   376→      ctx.fillStyle = '#ffffff';
   377→   377→      const eyeOff = enemy.vx > 0 ? 4 : -1;
   378→   378→      ctx.fillRect(enemy.x + enemy.width / 2 + eyeOff - 2, enemy.y + 6, 3, 3);
   379→   379→      ctx.fillRect(enemy.x + enemy.width / 2 + eyeOff + 4, enemy.y + 6, 3, 3);
   380→   380→    }
   381→   381→    ctx.shadowBlur = 0;
   382→   382→  }
   383→   383→
   384→   384→  // === Player ===
   385→   385→  if (player.isAlive) {
   386→   386→    const px = player.x;
   387→   387→    const py = player.y;
   388→   388→    const pw = player.width;
   389→   389→    const ph = player.height;
   390→   390→
   391→   391→    // Trail
   392→   392→    if (Math.abs(player.vx) > 1 || !player.onGround) {
   393→   393→      ctx.fillStyle = COLORS.playerTrail;
   394→   394→      ctx.beginPath();
   395→   395→      ctx.roundRect(px - 2, py + 4, pw, ph - 4, 4);
   396→   396→      ctx.fill();
   397→   397→    }
   398→   398→
   399→   399→    if (assets?.characterImg && assets.characterImg.complete) {
   400→   400→      // Character image sprite
   401→   401→      ctx.save();
   402→   402→      ctx.shadowColor = COLORS.playerGlow;
   403→   403→      ctx.shadowBlur = 12;
   404→   404→
   405→   405→      if (player.facing === 'left') {
   406→   406→        ctx.translate(px + pw / 2, 0);
   407→   407→        ctx.scale(-1, 1);
   408→   408→        ctx.drawImage(assets.characterImg, -pw / 2 - 2, py - 2, pw + 4, ph + 4);
   409→   409→      } else {
   410→   410→        ctx.drawImage(assets.characterImg, px - 2, py - 2, pw + 4, ph + 4);
   411→   411→      }
   412→   412→
   413→   413→      // Lightning aura
   414→   414→      const pulseAlpha = 0.12 + 0.08 * Math.sin(time / 300);
   415→   415→      ctx.strokeStyle = `rgba(0, 255, 170, ${pulseAlpha})`;
   416→   416→      ctx.lineWidth = 2;
   417→   417→      ctx.beginPath();
   418→   418→      ctx.roundRect(px - 4, py - 4, pw + 8, ph + 8, 8);
   419→   419→      ctx.stroke();
   420→   420→      ctx.restore();
   421→   421→    } else {
   422→   422→      // Fallback geometric player
   423→   423→      ctx.shadowColor = COLORS.playerGlow;
   424→   424→      ctx.shadowBlur = 12;
   425→   425→      ctx.fillStyle = COLORS.player;
   426→   426→      ctx.beginPath();
   427→   427→      ctx.roundRect(px, py, pw, ph, 5);
   428→   428→      ctx.fill();
   429→   429→
   430→   430→      const faceDir = player.facing === 'right' ? 1 : -1;
   431→   431→      ctx.fillStyle = '#0a0a2e';
   432→   432→      ctx.beginPath();
   433→   433→      ctx.arc(px + pw / 2 + faceDir * 4, py + 10, 3, 0, Math.PI * 2);
   434→   434→      ctx.fill();
   435→   435→      ctx.fillStyle = '#ffffff';

   512→   512→function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   513→   513→  ctx.beginPath();
   514→   514→  ctx.moveTo(x + r, y);
   515→   515→  ctx.lineTo(x + w - r, y);
   516→   516→  ctx.quadraticCurveTo(x + w, y, x + w, y + r);
   517→   517→  ctx.lineTo(x + w, y + h - r);
   518→   518→  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
   519→   519→  ctx.lineTo(x + r, y + h);
   520→   520→  ctx.quadraticCurveTo(x, y + h, x, y + h - r);
   521→   521→  ctx.lineTo(x, y + r);
   522→   522→  ctx.quadraticCurveTo(x, y, x + r, y);
   523→   523→  ctx.closePath();
   524→   524→}
